The Modern API is the global `window.dsx`, so there is nothing to install. Copy this service into your project.

## The service

Answers and broadcasts come from native code, outside Angular's zone. The service re-enters the zone with `NgZone.run`, so change detection sees them.

```typescript despia.service.ts
import { Injectable, NgZone, inject } from '@angular/core'

@Injectable({ providedIn: 'root' })
export class DespiaService {
  private zone = inject(NgZone)

  has(pkg: string): boolean {
    return !!window.dsx?.has(pkg)
  }

  // call('revenuecat', 'purchase', args): resolves with the result, rejects with { code, message }
  call<T = unknown>(pkg: string, action: string, args: object = {}): Promise<T> {
    const dsx = window.dsx
    if (!dsx) return Promise.reject({ code: 'no_dsx' }) // our own code: not in Despia V4
    const fn = action.split('.').reduce((m: any, key) => m[key], dsx.module[pkg])
    return new Promise<T>((resolve, reject) => {
      fn(args).then(
        (value: T) => this.zone.run(() => resolve(value)),
        (error: unknown) => this.zone.run(() => reject(error)),
      )
    })
  }

  // on('focus', 'change', handler): returns the function that stops listening
  on(pkg: string, name: string, handler: (data: any) => void): () => void {
    const dsx = window.dsx
    return dsx ? dsx.module[pkg].on(name, (data: any) => this.zone.run(() => handler(data))) : () => {}
  }
}
```

## Call a package and listen to a broadcast

```typescript buy-button.component.ts
import { Component, Input, OnDestroy, afterNextRender, inject } from '@angular/core'
import { DespiaService } from './despia.service'

@Component({
  selector: 'app-buy-button',
  standalone: true,
  template: `
    @if (ready) {
      <button [disabled]="loading" (click)="buy()">
        {{ loading ? 'Buying…' : 'Buy 100 coins' }}
        @if (error) { <small> ({{ error.code }})</small> }
      </button>
    }
  `,
})
export class BuyButtonComponent implements OnDestroy {
  @Input() userId = ''
  private despia = inject(DespiaService)
  private off: () => void = () => {}
  ready = false
  loading = false
  error: { code: string } | null = null

  constructor() {
    afterNextRender(() => { // runs only in the browser
      this.ready = this.despia.has('revenuecat')
      this.off = this.despia.on('focus', 'change', ({ focused }) => {
        if (!focused) console.log('app went to the background')
      })
    })
  }

  ngOnDestroy() {
    this.off()
  }

  async buy() {
    this.loading = true
    this.error = null
    try {
      await this.despia.call('revenuecat', 'purchase', { external_id: this.userId, product: ['coins_100'] })
    } catch (e: any) {
      this.error = e
    } finally {
      this.loading = false
    }
  }
}
```

## Angular SSR

Despia code runs in the browser only. With Angular SSR, use the service from event handlers or from `afterNextRender`, which only run in the browser, not from code that also runs on the server.
