The Modern API is the global `window.dsx`, so there is nothing to install. Copy these two hooks into your project.

## The hooks

```jsx despia-hooks.js
import { useCallback, useEffect, useRef, useState } from 'react'

const getDsx = () => window.dsx

// useDsxCall('revenuecat', 'purchase') -> { call, data, error, loading }
export function useDsxCall(pkg, action) {
  const [state, setState] = useState({ data: null, error: null, loading: false })
  const call = useCallback(async (args) => {
    const dsx = getDsx()
    if (!dsx) throw { code: 'no_dsx' } // our own code: not in Despia V4
    setState((s) => ({ ...s, error: null, loading: true }))
    try {
      const fn = action.split('.').reduce((m, key) => m[key], dsx.module[pkg])
      const data = await fn(args)
      setState({ data, error: null, loading: false })
      return data
    } catch (error) {
      setState({ data: null, error, loading: false })
      throw error
    }
  }, [pkg, action])
  return { ...state, call }
}

// useDsxBroadcast('focus', 'change', (data) => ...), removed on unmount
export function useDsxBroadcast(pkg, name, handler) {
  const latest = useRef(handler)
  latest.current = handler
  useEffect(() => {
    const dsx = getDsx()
    return dsx ? dsx.module[pkg].on(name, (data) => latest.current(data)) : undefined
  }, [pkg, name])
}
```

A failed call rejects with an object that has a `code` (for RevenueCat, for example `user_cancelled` or `not_ready`); `no_dsx` is the hook's own code for "not in Despia V4". `action` may contain dots, for example `'permission.request'` on `notify`.

## Call a package

```jsx BuyButton.jsx
import { useEffect, useState } from 'react'
import { useDsxCall } from './despia-hooks'

export function BuyButton({ userId }) {
  const [ready, setReady] = useState(false)
  const { call, loading, error } = useDsxCall('revenuecat', 'purchase')
  useEffect(() => setReady(!!window.dsx?.has('revenuecat')), [])

  if (!ready) return null // web checkout goes here

  return (
    <button disabled={loading} onClick={() => call({ external_id: userId, product: ['coins_100'] }).catch(() => {})}>
      {loading ? 'Buying…' : 'Buy 100 coins'}
      {error && <small> ({error.code})</small>}
    </button>
  )
}
```

`ready` is read in an effect, so the server render and the first client render are the same.

## Listen to a broadcast

```jsx
useDsxBroadcast('focus', 'change', ({ focused }) => { if (!focused) saveDraft() })
useDsxBroadcast('revenuecat', 'customerInfo', ({ customerInfo }) => refresh(customerInfo))
```

For a stream, return its `stop` from an effect:

```jsx
useEffect(() => {
  if (!window.dsx?.has('speechrecognition')) return undefined
  const sub = window.dsx.module.speechrecognition.start({ interim: true }, (e) => setText(e.data?.transcript))
  return () => sub.stop()
}, [])
```

## Next.js

Use the hooks in client components (`'use client'` in the App Router). They read `window.dsx` only inside effects and handlers, never during render.

```jsx app/buy/page.jsx
'use client'
import { BuyButton } from '../../components/BuyButton'

export default function Page() {
  return <BuyButton userId="user_123" />
}
```
