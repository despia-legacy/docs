The Modern API is the global `window.dsx`, so there is nothing to install. Copy these two primitives into your project.

## The primitives

```jsx despia.js
import { createSignal, onCleanup, onMount } from 'solid-js'

// createDsxCall('revenuecat', 'purchase') -> { call, data, error, loading }
export function createDsxCall(pkg, action) {
  const [data, setData] = createSignal(null)
  const [error, setError] = createSignal(null)
  const [loading, setLoading] = createSignal(false)
  async function call(args) {
    const dsx = window.dsx
    if (!dsx) throw setError({ code: 'no_dsx' }) // our own code: not in Despia V4
    setLoading(true)
    setError(null)
    try {
      const fn = action.split('.').reduce((m, key) => m[key], dsx.module[pkg])
      const result = await fn(args)
      setData(() => result)
      return result
    } catch (e) {
      setError(e)
      throw e
    } finally {
      setLoading(false)
    }
  }
  return { call, data, error, loading }
}

// onDsxBroadcast('focus', 'change', (data) => ...), removed on cleanup
export function onDsxBroadcast(pkg, name, handler) {
  onMount(() => {
    const dsx = window.dsx
    if (dsx) onCleanup(dsx.module[pkg].on(name, handler))
  })
}
```

## Call a package and listen to a broadcast

```jsx BuyButton.jsx
import { createSignal, onMount, Show } from 'solid-js'
import { createDsxCall, onDsxBroadcast } from './despia'

export function BuyButton(props) {
  const [ready, setReady] = createSignal(false)
  const { call, loading, error } = createDsxCall('revenuecat', 'purchase')
  onMount(() => setReady(!!window.dsx?.has('revenuecat')))
  onDsxBroadcast('focus', 'change', ({ focused }) => { if (!focused) console.log('app went to the background') })

  const buy = () => call({ external_id: props.userId, product: ['coins_100'] }).catch(() => {})

  return (
    <Show when={ready()}>
      <button disabled={loading()} onClick={buy}>
        {loading() ? 'Buying…' : 'Buy 100 coins'}
        <Show when={error()}>{(e) => <small> ({e().code})</small>}</Show>
      </button>
    </Show>
  )
}
```

## SolidStart

`onMount` runs only in the browser, so the primitives are safe with SolidStart. Never read `window.dsx` in a component body or a server function.
